```python
import numpy as np
import pandas as pd
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import f1_score
from sklearn.ensemble import HistGradientBoostingClassifier
import lightgbm as lgb

SEED = 42

train = pd.read_csv("train.csv")
test = pd.read_csv("test_features.csv")

target = "Revenue"
y = train[target].astype(int).values
X = train.drop(columns=[target])
Xt = test[X.columns.tolist()].copy()

months = ["Jan", "Feb", "Mar", "Apr", "May", "June", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"]
month_map = {m: i + 1 for i, m in enumerate(months)}

def prep(df):
    df = df.copy()
    df["Month_num"] = df["Month"].map(month_map)
    df["Month"] = df["Month"].astype("category")
    df["VisitorType"] = df["VisitorType"].astype("category")
    df["n_missing"] = df.isna().sum(axis=1)
    pr = df["ProductRelated"]
    df["pr_dur_per_page"] = df["ProductRelated_Duration"] / (pr + 1)
    df["total_pages"] = df["Administrative"] + df["Informational"] + pr
    df["total_dur"] = (df["Administrative_Duration"] + df["Informational_Duration"]
                       + df["ProductRelated_Duration"])
    df["pv_pos"] = (df["PageValues"] > 0).astype(float)
    df.loc[df["PageValues"].isna(), "pv_pos"] = np.nan
    df["exit_bounce"] = df["ExitRates"] - df["BounceRates"]
    return df

all_df = pd.concat([X, Xt], axis=0, ignore_index=True)
all_df = prep(all_df)
# consistent categories
for c in ["Month", "VisitorType"]:
    all_df[c] = all_df[c].astype(str).astype("category")
Xp = all_df.iloc[:len(X)].reset_index(drop=True)
Xtp = all_df.iloc[len(X):].reset_index(drop=True)

params = dict(
    n_estimators=400, learning_rate=0.03, num_leaves=15, min_child_samples=30,
    subsample=0.8, subsample_freq=1, colsample_bytree=0.7,
    reg_lambda=2.0, random_state=SEED, verbose=-1, n_jobs=4,
)

skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=SEED)
oof = np.zeros(len(Xp))
test_pred = np.zeros(len(Xtp))

for tr, va in skf.split(Xp, y):
    m = lgb.LGBMClassifier(**params)
    m.fit(Xp.iloc[tr], y[tr])
    oof[va] += m.predict_proba(Xp.iloc[va])[:, 1] * 0.5
    test_pred += m.predict_proba(Xtp)[:, 1] / 5

    h = HistGradientBoostingClassifier(
        learning_rate=0.05, max_iter=250, max_leaf_nodes=15,
        min_samples_leaf=30, l2_regularization=1.0, random_state=SEED,
        categorical_features="from_dtype",
    )
    h.fit(Xp.iloc[tr], y[tr])
    oof[va] += h.predict_proba(Xp.iloc[va])[:, 1] * 0.5
    test_pred += h.predict_proba(Xtp)[:, 1] / 5

# note: test_pred accumulated two models each /5 -> divide by 2
test_pred /= 2

# threshold tuning on OOF
best_t, best_f = 0.5, -1
for t in np.arange(0.15, 0.7, 0.01):
    f = f1_score(y, (oof >= t).astype(int))
    if f > best_f:
        best_f, best_t = f, t
print("best threshold", best_t, "oof F1", best_f)

out = pd.DataFrame({
    "proba": np.clip(test_pred, 0, 1),
    "label": (test_pred >= best_t).astype(int),
})
out.to_csv("predictions.csv", index=False)
```
