- **Missing values:** About 20% of each numeric column is missing, so I add a missing-count feature. The tree models then handle the NaNs natively, and the logistic regression gets median imputation.
- **Features:** I add a few derived ones: total pages and duration, average time per page, and the ratio of PageValues to ProductRelated. Month and VisitorType are one-hot encoded.
- **Models:** I blend LightGBM, XGBoost, CatBoost and a logistic regression into one probability.
- **Threshold:** I get out-of-fold predictions with 5-fold stratified CV, then pick the probability threshold that maximizes F1 on those predictions. The final models are refit on all the training data to predict the test set.
- **Runtime:** The models are small, so it should finish well within 15 minutes.
- **Output:** The script writes `proba` and `label` to `predictions.csv` in the same row order as `test_features.csv`.

```python
import numpy as np
import pandas as pd
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import f1_score
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.impute import SimpleImputer
from sklearn.pipeline import make_pipeline
import lightgbm as lgb
import xgboost as xgb
from catboost import CatBoostClassifier

train = pd.read_csv("train.csv")
test = pd.read_csv("test_features.csv")

y = train["Revenue"].astype(int).values
X = train.drop(columns=["Revenue"])
T = test.copy()

num_cols = [c for c in X.columns if c not in ("Month", "VisitorType")]


def fe(df):
    df = df.copy()
    df["n_missing"] = df[num_cols].isna().sum(axis=1)
    df["tot_pages"] = df[["Administrative", "Informational", "ProductRelated"]].sum(axis=1, min_count=1)
    df["tot_dur"] = df[["Administrative_Duration", "Informational_Duration",
                        "ProductRelated_Duration"]].sum(axis=1, min_count=1)
    df["dur_per_page"] = df["tot_dur"] / (df["tot_pages"] + 1)
    df["pr_dur_per_page"] = df["ProductRelated_Duration"] / (df["ProductRelated"] + 1)
    df["pv_per_pr"] = df["PageValues"] / (df["ProductRelated"] + 1)
    df["pv_pos"] = (df["PageValues"] > 0).astype(float).where(df["PageValues"].notna())
    df["exit_bounce"] = df["ExitRates"] - df["BounceRates"]
    months = ["Feb", "Mar", "May", "June", "Jul", "Aug", "Sep", "Oct", "Nov", "Dec"]
    for m in months:
        df["Month_" + m] = (df["Month"] == m).astype(int)
    for v in ["Returning_Visitor", "New_Visitor", "Other"]:
        df["VT_" + v] = (df["VisitorType"] == v).astype(int)
    df = df.drop(columns=["Month", "VisitorType"])
    return df


Xf = fe(X)
Tf = fe(T)
Tf = Tf[Xf.columns]


def make_models():
    return {
        "lgb": lgb.LGBMClassifier(n_estimators=400, learning_rate=0.03, num_leaves=15,
                                  min_child_samples=20, subsample=0.8, subsample_freq=1,
                                  colsample_bytree=0.8, reg_lambda=1.0,
                                  random_state=42, verbose=-1, n_jobs=-1),
        "xgb": xgb.XGBClassifier(n_estimators=400, learning_rate=0.03, max_depth=4,
                                 subsample=0.8, colsample_bytree=0.8, min_child_weight=3,
                                 eval_metric="logloss", random_state=42, n_jobs=-1,
                                 tree_method="hist"),
        "cat": CatBoostClassifier(iterations=600, learning_rate=0.04, depth=6,
                                  random_seed=42, verbose=0, thread_count=-1),
        "lr": make_pipeline(SimpleImputer(strategy="median"), StandardScaler(),
                            LogisticRegression(C=0.5, max_iter=2000, random_state=42)),
    }


weights = {"lgb": 1.0, "xgb": 1.0, "cat": 1.5, "lr": 0.5}


def blend(preds):
    s = sum(weights.values())
    return sum(weights[k] * preds[k] for k in preds) / s


skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
oof = {k: np.zeros(len(Xf)) for k in weights}
for tr, va in skf.split(Xf, y):
    ms = make_models()
    for k, m in ms.items():
        m.fit(Xf.iloc[tr], y[tr])
        oof[k][va] = m.predict_proba(Xf.iloc[va])[:, 1]

oof_blend = blend(oof)
best_t, best_f = 0.5, -1
for t in np.arange(0.1, 0.9, 0.01):
    f = f1_score(y, (oof_blend >= t).astype(int))
    if f > best_f:
        best_f, best_t = f, t
print("OOF F1: %.4f at threshold %.2f" % (best_f, best_t))

ms = make_models()
tp = {}
for k, m in ms.items():
    m.fit(Xf, y)
    tp[k] = m.predict_proba(Tf)[:, 1]
proba = np.clip(blend(tp), 0, 1)
label = (proba >= best_t).astype(int)

pd.DataFrame({"proba": proba, "label": label}).to_csv("predictions.csv", index=False)
```
