```python
import numpy as np
import pandas as pd
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import f1_score
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import make_pipeline
from sklearn.impute import SimpleImputer
import lightgbm as lgb

TARGET = "default.payment.next.month"
train = pd.read_csv("train.csv")
test = pd.read_csv("test_features.csv")

y = train[TARGET].astype(int).values
Xtr = train.drop(columns=[TARGET])
Xte = test[Xtr.columns]

pay_cols = ["PAY_0", "PAY_2", "PAY_3", "PAY_4", "PAY_5", "PAY_6"]
bill_cols = [f"BILL_AMT{i}" for i in range(1, 7)]
amt_cols = [f"PAY_AMT{i}" for i in range(1, 7)]


def fe(df):
    d = df.copy()
    d["pay_mean"] = d[pay_cols].mean(axis=1)
    d["pay_max"] = d[pay_cols].max(axis=1)
    d["pay_min"] = d[pay_cols].min(axis=1)
    d["pay_sum"] = d[pay_cols].sum(axis=1)
    d["n_delay"] = (d[pay_cols] > 0).sum(axis=1)
    d["bill_mean"] = d[bill_cols].mean(axis=1)
    d["bill_max"] = d[bill_cols].max(axis=1)
    d["amt_mean"] = d[amt_cols].mean(axis=1)
    d["amt_sum"] = d[amt_cols].sum(axis=1)
    d["util"] = d["bill_mean"] / (d["LIMIT_BAL"] + 1)
    for i in range(1, 7):
        d[f"util{i}"] = d[f"BILL_AMT{i}"] / (d["LIMIT_BAL"] + 1)
        d[f"ratio{i}"] = d[f"PAY_AMT{i}"] / (d[f"BILL_AMT{i}"].abs() + 1)
    d["pay_ratio"] = d["amt_sum"] / (d[bill_cols].sum(axis=1).abs() + 1)
    d["bill_trend"] = d["BILL_AMT1"] - d["BILL_AMT6"]
    d["n_missing"] = df.isna().sum(axis=1)
    return d


Xtr = fe(Xtr)
Xte = fe(Xte)

params = dict(
    n_estimators=400, learning_rate=0.02, num_leaves=15, min_child_samples=40,
    subsample=0.8, subsample_freq=1, colsample_bytree=0.6, reg_lambda=5.0,
    random_state=42, verbose=-1, n_jobs=-1,
)

skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
oof = np.zeros(len(Xtr))
pred = np.zeros(len(Xte))

lr_oof = np.zeros(len(Xtr))
lr_pred = np.zeros(len(Xte))

for tr, va in skf.split(Xtr, y):
    m = lgb.LGBMClassifier(**params)
    m.fit(Xtr.iloc[tr], y[tr])
    oof[va] = m.predict_proba(Xtr.iloc[va])[:, 1]
    pred += m.predict_proba(Xte)[:, 1] / 5

    # logistic regression with signed log transform of amounts
    def prep(d):
        d = d.copy()
        for c in bill_cols + amt_cols + ["bill_mean", "bill_max", "amt_mean", "amt_sum", "bill_trend"]:
            d[c] = np.sign(d[c]) * np.log1p(d[c].abs())
        d = d.replace([np.inf, -np.inf], np.nan)
        return d
    lr = make_pipeline(SimpleImputer(strategy="median"), StandardScaler(),
                       LogisticRegression(C=0.1, max_iter=2000))
    lr.fit(prep(Xtr.iloc[tr]), y[tr])
    lr_oof[va] = lr.predict_proba(prep(Xtr.iloc[va]))[:, 1]
    lr_pred += lr.predict_proba(prep(Xte))[:, 1] / 5

# blend choice
best_w, best_f, best_t = 1.0, -1, 0.5
ths = np.arange(0.15, 0.7, 0.01)
for w in [1.0, 0.8, 0.6]:
    b = w * oof + (1 - w) * lr_oof
    for t in ths:
        f = f1_score(y, (b >= t).astype(int))
        if f > best_f:
            best_f, best_w, best_t = f, w, t

proba = best_w * pred + (1 - best_w) * lr_pred
label = (proba >= best_t).astype(int)

pd.DataFrame({"proba": proba, "label": label}).to_csv("predictions.csv", index=False)
```
