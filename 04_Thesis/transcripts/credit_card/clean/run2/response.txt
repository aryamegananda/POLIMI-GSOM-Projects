```python
import numpy as np
import pandas as pd
from sklearn.model_selection import StratifiedKFold
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import make_pipeline
from sklearn.metrics import f1_score
import lightgbm as lgb

TARGET = "default.payment.next.month"
train = pd.read_csv("train.csv")
test = pd.read_csv("test_features.csv")


def fe(df):
    d = df.drop(columns=[TARGET], errors="ignore").copy()
    pays = ["PAY_0", "PAY_2", "PAY_3", "PAY_4", "PAY_5", "PAY_6"]
    bills = [f"BILL_AMT{i}" for i in range(1, 7)]
    pamts = [f"PAY_AMT{i}" for i in range(1, 7)]
    d["PAY_MEAN"] = d[pays].mean(axis=1)
    d["PAY_MAX"] = d[pays].max(axis=1)
    d["PAY_SUM_POS"] = d[pays].clip(lower=0).sum(axis=1)
    d["N_DELAY"] = (d[pays] > 0).sum(axis=1)
    d["BILL_MEAN"] = d[bills].mean(axis=1)
    d["PAYAMT_MEAN"] = d[pamts].mean(axis=1)
    d["PAYAMT_SUM"] = d[pamts].sum(axis=1)
    d["UTIL1"] = d["BILL_AMT1"] / d["LIMIT_BAL"]
    d["UTIL_MEAN"] = d["BILL_MEAN"] / d["LIMIT_BAL"]
    d["PAY_RATIO1"] = d["PAY_AMT1"] / (d["BILL_AMT2"].abs() + 1)
    d["PAY_RATIO_ALL"] = d["PAYAMT_SUM"] / (d[bills[1:]].sum(axis=1).abs() + 1)
    d["BILL_TREND"] = d["BILL_AMT1"] - d["BILL_AMT6"]
    d["PAYLIM"] = d["PAYAMT_MEAN"] / d["LIMIT_BAL"]
    return d


X = fe(train)
y = train[TARGET].values
Xt = fe(test)

lgb_params = dict(n_estimators=400, learning_rate=0.02, num_leaves=15,
                  min_child_samples=40, subsample=0.8, subsample_freq=1,
                  colsample_bytree=0.7, reg_lambda=5.0, random_state=42,
                  verbose=-1, n_jobs=-1)


def make_lr():
    return make_pipeline(StandardScaler(),
                         LogisticRegression(C=0.5, max_iter=2000, random_state=42))


def signed_log(df):
    return np.sign(df) * np.log1p(np.abs(df))


Xl, Xtl = signed_log(X), signed_log(Xt)

skf = StratifiedKFold(5, shuffle=True, random_state=42)
oof = np.zeros(len(X))
pt = np.zeros(len(Xt))
for tr, va in skf.split(X, y):
    m = lgb.LGBMClassifier(**lgb_params).fit(X.iloc[tr], y[tr])
    l = make_lr().fit(Xl.iloc[tr], y[tr])
    oof[va] = 0.75 * m.predict_proba(X.iloc[va])[:, 1] + 0.25 * l.predict_proba(Xl.iloc[va])[:, 1]
    pt += (0.75 * m.predict_proba(Xt)[:, 1] + 0.25 * l.predict_proba(Xtl)[:, 1]) / 5

ths = np.arange(0.15, 0.7, 0.01)
f1s = [f1_score(y, (oof > t).astype(int)) for t in ths]
best = ths[int(np.argmax(f1s))]
print("best threshold", best, "oof F1", max(f1s))

pd.DataFrame({"proba": np.clip(pt, 0, 1),
              "label": (pt > best).astype(int)}).to_csv("predictions.csv", index=False)
```
