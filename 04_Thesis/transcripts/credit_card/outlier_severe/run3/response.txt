The data looks like the UCI credit default set with injected noise: negative LIMIT_BAL, AGE of -1, and odd float values in some bill and payment columns. The script therefore clips and cleans values and adds engineered features. It trains gradient boosting models with out-of-fold predictions, blends them, and tunes the F1 threshold on the OOF blend.

Because it runs once with no feedback, I kept the code defensive. Models are LightGBM, CatBoost and logistic regression, evaluated with 5-fold stratified CV. The blend uses a simple average of the tree models, and the threshold is chosen by a grid search on the OOF probabilities.

```python
import pandas as pd
import numpy as np
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import f1_score
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import make_pipeline
import lightgbm as lgb
from catboost import CatBoostClassifier

T = "default.payment.next.month"
train = pd.read_csv("train.csv")
test = pd.read_csv("test_features.csv")

def fe(df):
    d = df.copy()
    d["LIMIT_BAL"] = d["LIMIT_BAL"].where(d["LIMIT_BAL"] > 0, np.nan)
    d["AGE"] = d["AGE"].where((d["AGE"] >= 18) & (d["AGE"] <= 100), np.nan)
    d["EDUCATION"] = d["EDUCATION"].replace({0: 4, 5: 4, 6: 4})
    d["MARRIAGE"] = d["MARRIAGE"].replace({0: 3})
    pays = ["PAY_0", "PAY_2", "PAY_3", "PAY_4", "PAY_5", "PAY_6"]
    bills = [f"BILL_AMT{i}" for i in range(1, 7)]
    pamt = [f"PAY_AMT{i}" for i in range(1, 7)]
    d["pay_mean"] = d[pays].mean(axis=1)
    d["pay_max"] = d[pays].max(axis=1)
    d["pay_min"] = d[pays].min(axis=1)
    d["pay_sum_pos"] = d[pays].clip(lower=0).sum(axis=1)
    d["n_late"] = (d[pays] > 0).sum(axis=1)
    d["pay_trend"] = d["PAY_0"] - d["PAY_6"]
    d["bill_mean"] = d[bills].mean(axis=1)
    d["bill_max"] = d[bills].max(axis=1)
    d["pamt_mean"] = d[pamt].mean(axis=1)
    d["pamt_sum"] = d[pamt].sum(axis=1)
    d["bill_sum"] = d[bills].sum(axis=1)
    d["pay_ratio"] = d["pamt_sum"] / (d["bill_sum"].abs() + 1)
    lb = d["LIMIT_BAL"]
    for i in range(1, 7):
        d[f"util{i}"] = d[f"BILL_AMT{i}"] / lb
        d[f"ratio{i}"] = d[f"PAY_AMT{i}"] / (d[f"BILL_AMT{i}"].abs() + 1)
    d["util_mean"] = d["bill_mean"] / lb
    d["bill_trend"] = d["BILL_AMT1"] - d["BILL_AMT6"]
    d["pamt_lim"] = d["pamt_mean"] / lb
    d = d.replace([np.inf, -np.inf], np.nan)
    return d

Xtr = fe(train.drop(columns=[T]))
y = train[T].values
Xte = fe(test)
Xte = Xte[Xtr.columns]

Xtr_f = Xtr.fillna(Xtr.median())
Xte_f = Xte.fillna(Xtr.median())

skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
oof = {k: np.zeros(len(Xtr)) for k in ["lgb", "cat", "lr"]}
pred = {k: np.zeros(len(Xte)) for k in oof}

for tr, va in skf.split(Xtr, y):
    m = lgb.LGBMClassifier(n_estimators=400, learning_rate=0.02, num_leaves=15,
                           min_child_samples=40, subsample=0.8, subsample_freq=1,
                           colsample_bytree=0.6, reg_lambda=5, random_state=42,
                           verbose=-1, n_jobs=-1)
    m.fit(Xtr.iloc[tr], y[tr])
    oof["lgb"][va] = m.predict_proba(Xtr.iloc[va])[:, 1]
    pred["lgb"] += m.predict_proba(Xte)[:, 1] / 5

    c = CatBoostClassifier(iterations=600, learning_rate=0.04, depth=6,
                           random_seed=42, verbose=0, thread_count=-1)
    c.fit(Xtr.iloc[tr], y[tr])
    oof["cat"][va] = c.predict_proba(Xtr.iloc[va])[:, 1]
    pred["cat"] += c.predict_proba(Xte)[:, 1] / 5

    # signed-log transform for LR
    def tl(X):
        X = X.copy()
        return np.sign(X) * np.log1p(np.abs(X))
    l = make_pipeline(StandardScaler(), LogisticRegression(C=0.5, max_iter=2000, random_state=42))
    l.fit(tl(Xtr_f.iloc[tr]), y[tr])
    oof["lr"][va] = l.predict_proba(tl(Xtr_f.iloc[va]))[:, 1]
    pred["lr"] += l.predict_proba(tl(Xte_f))[:, 1] / 5

w = {"lgb": 0.4, "cat": 0.5, "lr": 0.1}
oof_b = sum(w[k] * oof[k] for k in w)
pred_b = sum(w[k] * pred[k] for k in w)

best_t, best_f = 0.5, 0
for t in np.arange(0.15, 0.7, 0.005):
    f = f1_score(y, (oof_b >= t).astype(int))
    if f > best_f:
        best_f, best_t = f, t
print("best threshold", best_t, "OOF F1", best_f)

pred_b = np.clip(pred_b, 0, 1)
out = pd.DataFrame({"proba": pred_b, "label": (pred_b >= best_t).astype(int)})
out.to_csv("predictions.csv", index=False)
```
