```python
import numpy as np
import pandas as pd
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import f1_score
from sklearn.linear_model import LogisticRegression
from sklearn.preprocessing import StandardScaler
from sklearn.pipeline import make_pipeline
import lightgbm as lgb
from catboost import CatBoostClassifier

TARGET = "default.payment.next.month"
train = pd.read_csv("train.csv")
test = pd.read_csv("test_features.csv")


def fe(df):
    d = df.copy()
    # clean obvious noise
    d["AGE"] = d["AGE"].where(d["AGE"] > 0, np.nan)
    d["LIMIT_BAL"] = d["LIMIT_BAL"].where(d["LIMIT_BAL"] > 0, np.nan)
    d["EDUCATION"] = d["EDUCATION"].replace({0: 4, 5: 4, 6: 4})
    d["MARRIAGE"] = d["MARRIAGE"].replace({0: 3})
    pays = ["PAY_0", "PAY_2", "PAY_3", "PAY_4", "PAY_5", "PAY_6"]
    bills = [f"BILL_AMT{i}" for i in range(1, 7)]
    amts = [f"PAY_AMT{i}" for i in range(1, 7)]
    d["pay_mean"] = d[pays].mean(axis=1)
    d["pay_max"] = d[pays].max(axis=1)
    d["pay_sum"] = d[pays].sum(axis=1)
    d["pay_pos_cnt"] = (d[pays] > 0).sum(axis=1)
    d["pay_recent"] = d["PAY_0"] + d["PAY_2"]
    d["bill_mean"] = d[bills].mean(axis=1)
    d["bill_sum"] = d[bills].sum(axis=1)
    d["amt_mean"] = d[amts].mean(axis=1)
    d["amt_sum"] = d[amts].sum(axis=1)
    d["util1"] = d["BILL_AMT1"] / d["LIMIT_BAL"]
    d["util_mean"] = d["bill_mean"] / d["LIMIT_BAL"]
    d["pay_ratio"] = d["amt_sum"] / (d["bill_sum"].abs() + 1)
    for i in range(1, 6):
        d[f"ratio{i}"] = d[f"PAY_AMT{i}"] / (d[f"BILL_AMT{i+1}"].abs() + 1)
    d["bill_trend"] = d["BILL_AMT1"] - d["BILL_AMT6"]
    d["unpaid1"] = d["BILL_AMT1"] - d["PAY_AMT1"]
    d["limit_minus_bill"] = d["LIMIT_BAL"] - d["BILL_AMT1"]
    return d


Xtr = fe(train.drop(columns=[TARGET]))
y = train[TARGET].values
Xte = fe(test)
Xtr_f = Xtr.fillna(Xtr.median())
Xte_f = Xte.fillna(Xtr.median())
Xtr_f = Xtr_f.replace([np.inf, -np.inf], 0)
Xte_f = Xte_f.replace([np.inf, -np.inf], 0)

skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
oof = {k: np.zeros(len(Xtr)) for k in ["lgb", "cat", "lr"]}
pte = {k: np.zeros(len(Xte)) for k in ["lgb", "cat", "lr"]}

for tr, va in skf.split(Xtr, y):
    m = lgb.LGBMClassifier(n_estimators=400, learning_rate=0.02, num_leaves=15,
                           min_child_samples=40, subsample=0.8, subsample_freq=1,
                           colsample_bytree=0.6, reg_lambda=5, random_state=42,
                           verbose=-1, n_jobs=-1)
    m.fit(Xtr.iloc[tr], y[tr])
    oof["lgb"][va] = m.predict_proba(Xtr.iloc[va])[:, 1]
    pte["lgb"] += m.predict_proba(Xte)[:, 1] / 5

    c = CatBoostClassifier(iterations=600, learning_rate=0.04, depth=6,
                           random_seed=42, verbose=0, thread_count=-1)
    c.fit(Xtr_f.iloc[tr], y[tr])
    oof["cat"][va] = c.predict_proba(Xtr_f.iloc[va])[:, 1]
    pte["cat"] += c.predict_proba(Xte_f)[:, 1] / 5

    # linear model on signed-log transformed features
    def sl(X):
        return np.sign(X) * np.log1p(np.abs(X))
    l = make_pipeline(StandardScaler(), LogisticRegression(C=0.1, max_iter=2000))
    l.fit(sl(Xtr_f.iloc[tr].values), y[tr])
    oof["lr"][va] = l.predict_proba(sl(Xtr_f.iloc[va].values))[:, 1]
    pte["lr"] += l.predict_proba(sl(Xte_f.values))[:, 1] / 5

w = {"lgb": 0.4, "cat": 0.5, "lr": 0.1}
oof_b = sum(w[k] * oof[k] for k in w)
te_b = sum(w[k] * pte[k] for k in w)

best_t, best_f = 0.5, 0
for t in np.arange(0.15, 0.65, 0.005):
    f = f1_score(y, (oof_b >= t).astype(int))
    if f > best_f:
        best_f, best_t = f, t
print("best threshold", best_t, "oof f1", best_f)

out = pd.DataFrame({"proba": np.clip(te_b, 0, 1),
                    "label": (te_b >= best_t).astype(int)})
out.to_csv("predictions.csv", index=False)
```
