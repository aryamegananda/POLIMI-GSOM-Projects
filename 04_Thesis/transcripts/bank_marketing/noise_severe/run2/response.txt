The script below has not been run, so I haven't seen its output or any errors. It works as follows:

- **Data:** this looks like the bank marketing dataset with `duration` removed, so I add a few engineered features.
- **Models:** it trains LightGBM and CatBoost.
- **Validation:** it uses 5-fold stratified out-of-fold (OOF) predictions to pick a blend and a probability threshold that maximizes F1.
- **Final predictions:** it averages the fold models' test predictions and applies that threshold.
- **Runtime:** the models are kept small so the whole script finishes well within the time limit.

```python
import numpy as np
import pandas as pd
from sklearn.model_selection import StratifiedKFold
from sklearn.metrics import f1_score
import lightgbm as lgb
from catboost import CatBoostClassifier

train = pd.read_csv("train.csv")
test = pd.read_csv("test_features.csv")

y = train["y"].values
X = train.drop(columns=["y"])
Xt = test.copy()

cat_cols = [c for c in X.columns if X[c].dtype == object]

def fe(df):
    df = df.copy()
    df["was_contacted"] = (df["pdays"] != 999).astype(int)
    df["pdays_c"] = df["pdays"].replace(999, -1)
    df["age_bin"] = pd.cut(df["age"], [0, 25, 35, 45, 55, 65, 120], labels=False)
    df["euri_x_emp"] = df["euribor3m"] * df["emp.var.rate"]
    df["conf_minus_price"] = df["cons.conf.idx"] - df["cons.price.idx"]
    return df

X = fe(X)
Xt = fe(Xt)

# LightGBM version: category dtype
X_l, Xt_l = X.copy(), Xt.copy()
for c in cat_cols:
    cats = pd.Categorical(pd.concat([X[c], Xt[c]])).categories
    X_l[c] = pd.Categorical(X[c], categories=cats)
    Xt_l[c] = pd.Categorical(Xt[c], categories=cats)

skf = StratifiedKFold(n_splits=5, shuffle=True, random_state=42)
oof_l = np.zeros(len(X)); oof_c = np.zeros(len(X))
te_l = np.zeros(len(Xt)); te_c = np.zeros(len(Xt))

for tr, va in skf.split(X, y):
    m = lgb.LGBMClassifier(n_estimators=2000, learning_rate=0.02, num_leaves=15,
                           min_child_samples=40, subsample=0.8, subsample_freq=1,
                           colsample_bytree=0.7, reg_lambda=5.0, cat_smooth=20,
                           random_state=42, verbose=-1, n_jobs=-1)
    m.fit(X_l.iloc[tr], y[tr], eval_set=[(X_l.iloc[va], y[va])],
          callbacks=[lgb.early_stopping(100, verbose=False)])
    oof_l[va] = m.predict_proba(X_l.iloc[va])[:, 1]
    te_l += m.predict_proba(Xt_l)[:, 1] / 5

    c = CatBoostClassifier(iterations=1500, learning_rate=0.04, depth=6,
                           random_seed=42, verbose=0, thread_count=-1,
                           cat_features=cat_cols, early_stopping_rounds=100)
    c.fit(X.iloc[tr], y[tr], eval_set=(X.iloc[va], y[va]))
    oof_c[va] = c.predict_proba(X.iloc[va])[:, 1]
    te_c += c.predict_proba(Xt)[:, 1] / 5

def best_thr(p):
    best, bt = 0, 0.5
    for t in np.arange(0.1, 0.8, 0.005):
        f = f1_score(y, (p >= t).astype(int))
        if f > best:
            best, bt = f, t
    return bt, best

best = (0, None, None)
for w in [0, 0.25, 0.5, 0.75, 1]:
    p = w * oof_l + (1 - w) * oof_c
    t, f = best_thr(p)
    if f > best[0]:
        best = (f, w, t)
f, w, t = best
print("OOF F1", f, "weight", w, "threshold", t)

proba = w * te_l + (1 - w) * te_c
label = (proba >= t).astype(int)
pd.DataFrame({"proba": proba, "label": label}).to_csv("predictions.csv", index=False)
```
